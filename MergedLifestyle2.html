# Define minimum valid pairs for LOESS
MIN_VALID_PAIRS <- 10  # Adjust as needed
LOESS_SPAN <- 0.75      # Smoothness parameter

# =============================================
# Step 2: Fit LOESS curves per disease group
# =============================================

loess_fits <- lifestyle_7var_ptau %>%
  filter(!(dx_entry == "MCI" |dx_entry == "CN" | dx_entry == "SMC")) %>%
  group_by(dx_entry) %>%
  group_modify(~ {
    # Extract data for the current group
    group_data <- .x
    age_vals <- group_data$age_at_csf_ptau
    ptau_vals <- group_data$ptau217_csf
    
    # Check for valid pairs
    valid_idx <- !is.na(age_vals) & !is.na(ptau_vals)
    n_valid <- sum(valid_idx)
    
    if (n_valid < MIN_VALID_PAIRS) {
      message(sprintf("Skipping %s: Only %d valid pairs (minimum %d required).",
                      .y$dx_entry, n_valid, MIN_VALID_PAIRS))
      return(tibble::tibble())  # Return empty tibble
    }
    
    fit_df <- data.frame(x = age_vals[valid_idx], y = ptau_vals[valid_idx])
    loess_fit <- tryCatch(
      loess(y ~ x, span = LOESS_SPAN, degree = 1, data = fit_df),
      error = function(e) NULL
    )
    if (is.null(loess_fit)) {
      message(sprintf("LOESS fitting failed for %s. Skipping.", .y$dx_entry))
      return(tibble::tibble())
    }
    age_seq <- seq(min(age_vals, na.rm = TRUE), max(age_vals, na.rm = TRUE), length.out = 100)
    fitted_vals <- predict(loess_fit, newdata = data.frame(x = age_seq))
    
    # Ensure fitted_vals has 100 rows
    if (length(fitted_vals) != 100) {
      stop("Prediction failed: fitted_vals does not have 100 rows.")
    }

    # Create a data frame for the current group
    tibble::tibble(
      age = age_seq,                           # 100 values
      fitted = fitted_vals                     # 100 values
    )
  }) %>%
  filter(n() > 0)  # Remove empty groups

# =============================================
# Step 3: Plot the LOESS curves
# =============================================

# Check if there are any groups left to plot
if (nrow(loess_fits) == 0) {
  stop("No groups had sufficient data for LOESS fitting. Check your dataset.")
}

# Create the plot
ggplot(lifestyle_7var_ptau, aes(x = age_at_csf_ptau, y = ptau217_csf, color = dx_entry)) +  # Raw data points
  geom_line(
    data = loess_fits,
    aes(x = age, y = fitted, group = dx_entry, color = dx_entry),
    linewidth = 1.2
  ) +  # LOESS curves
  labs(
    title = "PTAU217 CSF Levels by Disease Stage (LOESS Smoothing)",
    x = "Age at CSF PTAU Measurement",
    y = "PTAU217 CSF Levels",
    color = "Disease Stage"
  ) +
  theme_minimal() +
  theme(legend.position = "top")
